# BetaFlight 3D Edition v0.3.1-test: best 3D settings found so far (ZoeFPV, Pass 53/56)
# v0.3.1 ships these 3D values as its defaults. Paste this if you upgraded without a full chip erase, or to put them back
# after an A/B test. Not a full tune: PIDs, rates, receiver, OSD stay as they are.
# Dynamic idle and RPM release need bidirectional DShot RPM telemetry (dshot_bidir = ON: AM32, Bluejay, BLHeli_32).
# Without it, RPM release falls back to the 250 ms timer and dynamic idle stays off; leave the dyn_idle_* lines out.
# UNPLUG THE BATTERY FIRST (3D: saving with a battery in can spin the motors up). After saving, bench-check props OFF:
# ~10 fast reversals through center, all motors restart cleanly.

set 3d_iterm_mode = PER_DIRECTION
set 3d_iterm_release = RPM
set 3d_iterm_load = FLIP
set 3d_iterm_reset_ms = 250
set 3d_iterm_min_ms = 50
set 3d_iterm_max_ms = 400
set 3d_iterm_rpm_pct = 40
set 3d_iterm_learn_thr = 50
set 3d_iterm_learn_hz_x10 = 5
set 3d_iterm_axes = ALL
set 3d_anti_gravity = GATED
set 3d_ag_approach_pct = 15
set 3d_dyn_idle = ON

profile 0
# 50 (5000 rpm) for 1S/2S whoops with high-kv motors (flown); 30 for 5-inch / low-kv (log replay only, not flown yet).
# Keep it below the rpm of your slowest motor
# at center stick.
set dyn_idle_min_rpm = 50
set dyn_idle_max_increase = 100

save
