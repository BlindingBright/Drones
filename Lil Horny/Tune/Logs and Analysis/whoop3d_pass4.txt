# Pass 3 wobble (8-60 Hz tracking error, roll) sits at 12-15 Hz and grows with throttle:
#   10-20% 7.2 | 35-60% 5.0 | 60-100% 8.4   (largest at high throttle)
# TPA was cutting D by up to 65% exactly there. The RPM filter fix leaves D-term noise
# very low (-30 dB above 400 Hz), so we can afford to keep more D.
# In 3D the TPA throttle is measured per direction from center, so 1450 = 45% of
# upright OR inverted throttle (inverted hover sits ~44%, so it keeps full D now).
set tpa_rate = 45
set tpa_breakpoint = 1450

# Roll wobble at low forward throttle doubled vs Pass 1 (3.4 -> 7.2). More baseline roll D
# (pitch already got this in Pass 2: d_min_pitch 38).
set d_min_roll = 34

save
