# --- Sliders off: raw values below are the tune --------------------------------
set simplified_pids_mode = OFF
set simplified_dterm_filter = OFF
set simplified_gyro_filter = OFF

# --- Motors / 3D --------------------------------------------------------------
set motor_poles = 12
set dshot_idle_value = 200
# 2% (was 4%): AM32 min duty 3 adds on top, so effective idle is back near ~5% (Pass 12)
set thrust_linear = 10
# 10 (was 20): AM32 min duty 3 + KV 2100 now supply the low-end punch; halving this cut the
# post-reversal roll wobble (130-250 ms) from 20.9 to 9.1 deg/s (Pass 14)
# vbat_sag_compensation stays 0 on purpose: pilot uses the power fade as the land cue

# --- Filters (RPM filter on target thanks to motor_poles 12) -----------------
set gyro_lpf1_dyn_min_hz = 300
set gyro_lpf1_dyn_max_hz = 600
set dterm_lpf1_dyn_min_hz = 90
set dterm_lpf1_dyn_max_hz = 180
set dterm_lpf2_static_hz = 180

# --- PIDs ---------------------------------------------------------------------
set i_roll = 88
set f_roll = 110
set d_min_roll = 38
set d_roll = 44
# roll D raised to match pitch for post-reversal propwash (Pass 13)

set p_pitch = 44
set f_pitch = 100
set d_min_pitch = 42

set p_yaw = 50
set i_yaw = 95

# --- TPA (3D: measured from center in each direction) -------------------------
set tpa_rate = 45
set tpa_breakpoint = 1450

save
